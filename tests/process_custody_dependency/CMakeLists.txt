cmake_minimum_required(VERSION 3.21)
project(process_custody_dependency_contract LANGUAGES NONE)

set(BUILD_TESTING ON CACHE BOOL "Parent test policy")
set(FETCHCONTENT_FULLY_DISCONNECTED ON)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(fixture "${CMAKE_CURRENT_BINARY_DIR}/provider")
set(provider_mode "${MODE}")
set(has_deadline TRUE)
set(retires_output TRUE)
if(provider_mode MATCHES "^old-")
    set(has_deadline FALSE)
    string(REGEX REPLACE "^old-" "" provider_mode "${provider_mode}")
elseif(provider_mode MATCHES "^no-retirement-")
    set(retires_output FALSE)
    string(REGEX REPLACE "^no-retirement-" "" provider_mode "${provider_mode}")
endif()

# These compiler-free cases exercise target-platform decisions on every host.
set(WIN32 FALSE)
set(APPLE FALSE)
set(CMAKE_SYSTEM_NAME "${TEST_SYSTEM_NAME}")
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    set(APPLE TRUE)
endif()

if(provider_mode STREQUAL "existing")
    add_library(vnm_process_custody::vnm_process_custody INTERFACE IMPORTED GLOBAL)
    if(has_deadline)
        set_property(TARGET vnm_process_custody::vnm_process_custody PROPERTY
            VNM_PROCESS_CUSTODY_OWNER_START_DEADLINE 1)
    endif()
    if(retires_output)
        set_property(TARGET vnm_process_custody::vnm_process_custody PROPERTY
            VNM_PROCESS_CUSTODY_DARWIN_PTY_OUTPUT_RETIREMENT 1)
    endif()
    set(VNM_PROCESS_CUSTODY_SOURCE_DIR "${fixture}/missing")
elseif(provider_mode STREQUAL "installed")
    file(MAKE_DIRECTORY "${fixture}/package")
    set(installed_provider
        "add_library(vnm_process_custody::vnm_process_custody INTERFACE IMPORTED GLOBAL)\n")
    if(has_deadline)
        string(APPEND installed_provider
            "set_property(TARGET vnm_process_custody::vnm_process_custody PROPERTY VNM_PROCESS_CUSTODY_OWNER_START_DEADLINE 1)\n")
    endif()
    if(retires_output)
        string(APPEND installed_provider
            "set_property(TARGET vnm_process_custody::vnm_process_custody PROPERTY VNM_PROCESS_CUSTODY_DARWIN_PTY_OUTPUT_RETIREMENT 1)\n")
    endif()
    file(WRITE "${fixture}/package/vnm_process_custody-config.cmake" "${installed_provider}")
    set(vnm_process_custody_DIR "${fixture}/package")
elseif(provider_mode STREQUAL "invalid")
    file(MAKE_DIRECTORY "${fixture}")
    file(WRITE "${fixture}/CMakeLists.txt"
        "message(FATAL_ERROR \"The framework root must not be configured\")\n")
    set(VNM_PROCESS_CUSTODY_SOURCE_DIR "${fixture}")
elseif(provider_mode STREQUAL "source" OR provider_mode STREQUAL "fallback")
    file(MAKE_DIRECTORY "${fixture}/cpp/process_custody/include/vnm_process_custody")
    file(WRITE "${fixture}/CMakeLists.txt"
        "message(FATAL_ERROR \"The framework root must not be configured\")\n")
    file(WRITE "${fixture}/cpp/process_custody/include/vnm_process_custody/process_spawn.h" "")
    file(WRITE "${fixture}/cpp/process_custody/CMakeLists.txt"
        "add_library(vnm_process_custody INTERFACE)\n"
        "add_library(vnm_process_custody::vnm_process_custody ALIAS vnm_process_custody)\n")
    if(has_deadline)
        file(APPEND "${fixture}/cpp/process_custody/CMakeLists.txt"
            "set_property(TARGET vnm_process_custody PROPERTY VNM_PROCESS_CUSTODY_OWNER_START_DEADLINE 1)\n")
    endif()
    if(retires_output)
        file(APPEND "${fixture}/cpp/process_custody/CMakeLists.txt"
            "set_property(TARGET vnm_process_custody PROPERTY VNM_PROCESS_CUSTODY_DARWIN_PTY_OUTPUT_RETIREMENT 1)\n")
    endif()
    if(provider_mode STREQUAL "source")
        set(VNM_PROCESS_CUSTODY_SOURCE_DIR "${fixture}/cpp/process_custody")
    else()
        set(CMAKE_DISABLE_FIND_PACKAGE_vnm_process_custody ON)
        set(FETCHCONTENT_SOURCE_DIR_VNM_TERMINAL_PROCESS_CUSTODY "${fixture}/cpp/process_custody")
    endif()
else()
    message(FATAL_ERROR "Unknown resolver test mode: ${MODE}")
endif()

if(SURFACE_PACKAGE)
    include("${CMAKE_CURRENT_LIST_DIR}/surface_package.cmake")
else()
    include("${DEPENDENCY_MODULE}")
endif()
if(NOT TARGET vnm_process_custody::vnm_process_custody)
    message(FATAL_ERROR "The selected provider is unavailable")
endif()
get_property(parent_testing CACHE BUILD_TESTING PROPERTY VALUE)
if(NOT BUILD_TESTING OR NOT parent_testing)
    message(FATAL_ERROR "Dependency resolution changed the parent's testing policy")
endif()
if(TARGET Qt6::Core OR TARGET vnm_framework::vnm_framework)
    message(FATAL_ERROR "The native leaf resolver pulled in the framework or Qt")
endif()
